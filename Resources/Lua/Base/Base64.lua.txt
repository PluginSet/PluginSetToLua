local base64bit = require("Base.bit")

local Base64 = {}

function Base64.Init(origin_str)
	Base64.code = {}
	for i = 1, #origin_str, 1 do
		table.insert(Base64.code, string.byte(string.sub(origin_str, i, i)))
	end

	Base64.reverse = {}
	for i = 1, 128, 1 do
		Base64.reverse[i] = 0
	end

	for i = 1, #Base64.code, 1 do
		Base64.reverse[Base64.code[i]] = i - 1
	end
end

Base64.Init("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/")

function Base64.Encode( input )
	local input_len = #input
	local result_len = math.floor((input_len + 2) / 3) * 4
	local result = {}
	local index = 1
	local i = 3
	while i <= input_len do
		local tmp1 = base64bit:_lshift(input[i - 2], 16)
		local tmp2 = base64bit:_lshift(input[i - 1], 8)
		local val = base64bit:_or(base64bit:_or(tmp1, tmp2), input[i])
		result[index] = Base64.code[base64bit:_rshift(val, 18) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_and(base64bit:_rshift(val, 12), 63) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_and(base64bit:_rshift(val, 6), 63) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_and(val, 63) + 1]
		index = index + 1
		i = i + 3
	end
	i = i - 2

	local mod = input_len % 3
	if mod == 1 then
		result[index] = Base64.code[base64bit:_rshift(input[i], 2) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_and(base64bit:_lshift(input[i], 4), 63) + 1]
		index = index + 1
	elseif mod == 2 then
		result[index] = Base64.code[base64bit:_rshift(input[i], 2) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_or(base64bit:_and(base64bit:_lshift(input[i], 4), 63), base64bit:_rshift(input[i+1], 4)) + 1]
		index = index + 1
		result[index] = Base64.code[base64bit:_and(base64bit:_lshift(input[i+1], 2), 63) + 1]
		index = index + 1
	end

	while index <= result_len do
		result[index] = string.byte("=")
		index = index + 1
	end
	return result
end

function Base64.Decode( input )
	local input_len = #input
	local len = math.floor(input_len / 4) * 3
	local tail = 0
	if len > 0 then
		tail = tail + (input[input_len] == string.byte("=") and 1 or 0)
		tail = tail + (input[input_len - 1] == string.byte("=") and 1 or 0)
		len = len - tail
	end
	local result = {}
	local i, index, search_len = 3, 1, input_len - tail
	while i < search_len do
		local tmp1 = base64bit:_lshift(Base64.reverse[input[i-2]], 18)
		local tmp2 = base64bit:_lshift(Base64.reverse[input[i-1]], 12)
		local tmp3 = base64bit:_lshift(Base64.reverse[input[i]], 6)
		local val = base64bit:_or(base64bit:_or(base64bit:_or(tmp1, tmp2), tmp3), Base64.reverse[input[i + 1]])
		result[index] = base64bit:_rshift(val, 16)
		index = index + 1
		result[index] = base64bit:_and(base64bit:_rshift(val, 8), 255)
		index = index + 1
		result[index] = base64bit:_and(val, 255)
		index = index + 1
		i = i + 4
	end
	i = i - 3
	if tail == 1 then
		local tmp1 = base64bit:_lshift(Base64.reverse[input[i + 1]], 12)
		local tmp2 = base64bit:_lshift(Base64.reverse[input[i+2]], 6)
		local val = base64bit:_or(base64bit:_or(tmp1, tmp2), Base64.reverse[input[i+3]])
		result[index] = base64bit:_rshift(val, 10)
		index = index + 1
		result[index] = base64bit:_and(base64bit:_rshift(val, 2), 255)
		index = index + 1
	elseif tail == 2 then
		val = base64bit:_or(base64bit:_lshift(Base64.reverse[input[i+1]], 6), Base64.reverse[input[i+2]])
		result[index] = base64bit:_rshift(val, 4)
		index = index + 1
	end
	return result
end

function Base64.EncodeString( str )
	local input = {}
	for i = 1, #str, 1 do
		table.insert(input, string.byte(string.sub(str, i, i)))
	end
	local output = Base64.Encode(input)
	local result = ''
	for k,v in ipairs(output) do
		result = result .. string.char(v)
	end
	return result
end

function Base64.DecodeString( str )
	local input = {}
    for i = 1, #str, 1 do
        table.insert(input, string.byte(string.sub(str, i, i)))
    end
    local output = Base64.Decode( input )
    local result = ''
    for k,v in ipairs(output) do
        result = result .. string.char(v)
    end
    return result
	-- body
end
